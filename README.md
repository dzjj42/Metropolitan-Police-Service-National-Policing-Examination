      ],
      footer:{text:"Independent educational practice resource — not an official police promotion decision"},
      timestamp:new Date().toISOString()
    }]
  };
  try{
    const response = await fetch(webhookUrl.trim(),{
      method:"POST",
      headers:{"Content-Type":"application/json"},
      body:JSON.stringify(payload)
    });
    if(!response.ok) throw new Error("Discord returned HTTP "+response.status);
    status.textContent = "Discord result notification sent.";

    status.textContent = "Could not send Discord notification. Check the webhook URL and browser/network permissions.";
